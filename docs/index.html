Traditional Approach Boilerplate 
// Manual getter/setter boilerplate
val prefs = context.getSharedPreferences("user", MODE_PRIVATE)

var userToken: String?
    get() = prefs.getString("user_token", null)
    set(value) {
        prefs.edit().putString("user_token", value).apply()
    }

var userNickname: String
    get() = prefs.getString("user_nickname", null) ?: "unknown"
    set(value) {
        prefs.edit().putString("user_nickname", value).apply()
    }
With RePropertyX Clean & Type-Safe 
// Expressive 1-line delegates!
val prefs = context.getSharedPreferences("user", MODE_PRIVATE)

// 1-line nullable delegate bound to "user_token"
var userToken: String? by prefs.byString("user_token")

// 1-line non-null delegate with fallback default
var userNickname: String by prefs.byString("user_nickname")
    .orElse { "unknown" }
Traditional ViewAnimator Complex 
// Manual ValueAnimator management
var animator: ValueAnimator? = null

fun animateY(view: View, targetY: Float) {
    animator?.cancel()
    animator = ValueAnimator.ofFloat(view.translationY, targetY).apply {
        duration = 300
        interpolator = FastOutSlowInInterpolator()
        addUpdateListener {
            view.translationY = it.animatedValue as Float
        }
        start()
    }
}
With RePropertyX Declarative 
// Declarative property animation
var View.animatedY by propertyOf(
    get = { translationY },
    set = { translationY = it }
).animated()

fun animateY(view: View, targetY: Float) {
    view.animatedY = targetY // Smoothly animates to targetY!
}
Standard Compose State Disconnected 
// Manual SharedPreferences sync & effect
@Composable
fun PreferenceSwitch(prefs: SharedPreferences) {
    var darkMode by remember {
        mutableStateOf(prefs.getBoolean("dark_mode", false))
    }

    Switch(
        checked = darkMode,
        onCheckedChange = { newSetting ->
            darkMode = newSetting
            prefs.edit().putBoolean("dark_mode", newSetting).apply()
        }
    )
}
With RePropertyX Seamless 
// Auto-synced MutableState property delegate!
@Composable
fun PreferenceSwitch(prefs: SharedPreferences) {
    var darkMode by rememberPropertyState(
        disposable = prefs.changesComposed()
    ) { prefs.byBoolean("dark_mode").orElse { false } }

    Switch(
        checked = darkMode,
        onCheckedChange = { newSetting ->
            darkMode = newSetting // Updates Compose UI AND SharedPreferences!
        }
    )
}
Traditional Stack Management Boilerplate 
// Manual list stack & index tracking
private val history = mutableListOf<String>()
private var index = -1

fun updateText(value: String) {
    history.add(value)
    index++
}

fun undo(): String? {
    if (index > 0) index--
    return if (index >= 0) history[index] else null
}
With RePropertyX 1-Line Extension 
// 1-line Undo/Redo stack property wrapper!
val textProp = mutablePropertyOf("Hello").withHistory(maxSize = 10)
var text: String by textProp

text = "World"
textProp.undo() // Instantly reverts back to "Hello"!
textProp.redo() // Redoes change back to "World"!
Manual SavedStateHandle Repetitive 
// Manual SavedStateHandle get/set getters
class SearchViewModel(state: SavedStateHandle) : ViewModel() {
    var query: String
        get() = state.get<String>("query") ?: ""
        set(value) {
            state["query"] = value
        }
}
With RePropertyX Composable & Safe 
// Expressive SavedStateHandle delegate + Operators!
class SearchViewModel(state: SavedStateHandle) : ViewModel() {
    var query: String by state.byProperty("query", default = "")
        .distinctUntilChanged()
        .onEach { logSearch(it) }
}

Quick API Reference

Search and discover available property delegate operators.

🔍
propertyOf(get, set)
Create a property delegate from getter & setter functions.
.map(transform)
Transform values between exposed and stored types.
.distinctUntilChanged()
Prevent redundant property assignments if assigned value is unchanged.
.onEach { action }
Observe property value changes with side-effect callback.
.onEachBefore { action }
Execute side-effect callback before applying new property value.
.animated()
Animate View property assignments smoothly with ValueAnimator.
prefs.byString()
Type-safe SharedPreferences String delegate.
prefs.byInt()
Type-safe SharedPreferences Int delegate.
rememberProperty { delegate }
Remember a property delegate inside Jetpack Compose UI.
.closable()
Automatically close previous Closable resource upon assigning a new one.
.orElse { fallback }
Provide fallback default value when getter returns null.
mutablePropertyOf(initial)
Create a simple in-memory mutable property delegate.
byThreadLocal { factory }
Type-safe ThreadLocal property delegate for thread-bound state.
0.byAtomic() / byAtomic()
Thread-safe AtomicInteger, AtomicReference, etc. property delegates.
byDeclaredField { "_field" }
Reflection field delegate with optional field caching.
.switchMap { child }
Switch dynamically to child property delegates based on parent value.
StateFlow.getValue()
Coroutines StateFlow and MutableStateFlow property delegation.
view.animatedFloatAwait()
Suspend function animating View properties asynchronously in Coroutines.
prefs.changesComposed()
Auto-register listener to recompose Compose UI on SharedPreferences changes.
.validate { condition }
Validate values on read/write and throw exception if invalid.
.catch { handler }
Catch exceptions thrown during property read/write operations.
